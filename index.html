<script>

(function () {

  const WEDDING = new Date(2027, 5, 5); // 2027-06-05

  const STORE_KEY = 'wedding-todos-v1';

  const DOW = ['일', '월', '화', '수', '목', '금', '토'];


  let todos = JSON.parse(localStorage.getItem(STORE_KEY) || '[]');

  let viewYear = new Date().getFullYear();

  let viewMonth = new Date().getMonth();

  let selectedDate = fmt(new Date());

  let filter = 'all';

  let editingId = null;


  const $ = (id) => document.getElementById(id);

  const save = () => localStorage.setItem(STORE_KEY, JSON.stringify(todos));

  function fmt(d) {

    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');

  }

  const ownerName = (o) => (o === 'groom' ? '신랑' : '신부');

  const uid = () => Date.now().toString(36) + Math.random().toString(36).slice(2, 6);


  // D-day

  function renderDday() {

    const today = new Date(); today.setHours(0, 0, 0, 0);

    const diff = Math.round((WEDDING - today) / 86400000);

    $('dday').textContent = diff > 0 ? `웨딩데이까지 D-${diff}` : diff === 0 ? '오늘이 웨딩데이입니다! 💐' : `웨딩데이 D+${-diff}`;

  }


  // 캘린더

  function renderCalendar() {

    $('monthTitle').textContent = `${viewYear}년 ${viewMonth + 1}월`;

    $('dowRow').innerHTML = DOW.map(d => `<div class="dow">${d}</div>`).join('');

    const first = new Date(viewYear, viewMonth, 1);

    const start = new Date(viewYear, viewMonth, 1 - first.getDay());

    const todayStr = fmt(new Date());

    const weddingStr = fmt(WEDDING);

    let html = '';

    for (let i = 0; i < 42; i++) {

      const d = new Date(start.getFullYear(), start.getMonth(), start.getDate() + i);

      const ds = fmt(d);

      const dayTodos = todos.filter(t => t.date === ds);

      const classes = ['day'];

      if (d.getMonth() !== viewMonth) classes.push('other');

      if (ds === todayStr) classes.push('today');

      if (ds === selectedDate) classes.push('selected');

      if (ds === weddingStr) classes.push('wedding');

      const dots = dayTodos.slice(0, 6).map(t => `<span class="dot ${t.owner}${t.done ? ' done' : ''}"></span>`).join('');

      html += `<div class="${classes.join(' ')}" data-date="${ds}">

        <div class="num">${d.getDate()}</div>

        ${ds === weddingStr ? '<div class="wedding-tag">💍 WEDDING</div>' : ''}

        <div class="dots">${dots}</div>

      </div>`;

    }

    $('calGrid').innerHTML = html;

  }


  // 우측 패널 (월별 todo)

  function renderPanel() {

    const monthPrefix = `${viewYear}-${String(viewMonth + 1).padStart(2, '0')}`;

    const monthTodos = todos.filter(t => t.date.startsWith(monthPrefix));

    const shown = monthTodos.filter(t => filter === 'all' || t.owner === filter);


    $('panelTitle').textContent = `${viewMonth + 1}월 할 일`;

    $('panelSub').textContent = `${selectedDate} 기준 · 이번 달 ${monthTodos.length}개`;


    const doneCount = monthTodos.filter(t => t.done).length;

    const pct = monthTodos.length ? Math.round(doneCount / monthTodos.length * 100) : 0;

    $('progressBar').style.width = pct + '%';

    $('progressText').textContent = `완료 ${doneCount} / ${monthTodos.length} (${pct}%)`;


    document.querySelectorAll('.filters button').forEach(b => b.classList.toggle('active', b.dataset.filter === filter));


    const groups = [

      { key: 'groom', label: '🤵 신랑 담당', color: 'var(--groom)' },

      { key: 'bride', label: '👰 신부 담당', color: 'var(--bride)' },

    ].filter(g => filter === 'all' || filter === g.key);


    if (!shown.length) {

      $('todoList').innerHTML = `<div class="empty">이번 달 할 일이 없습니다.<br>위 폼에서 추가해 보세요.</div>`;

      return;

    }


    $('todoList').innerHTML = groups.map(g => {

      const items = shown.filter(t => t.owner === g.key).sort((a, b) => a.date.localeCompare(b.date) || (a.done - b.done));

      if (!items.length) return '';

      return `<div class="group-title"><i style="background:${g.color}"></i>${g.label} (${items.length})</div>` +

        items.map(t => `

          <div class="todo ${t.done ? 'done' : ''}">

            <input type="checkbox" data-act="toggle" data-id="${t.id}" ${t.done ? 'checked' : ''}>

            <div class="body">

              <div class="title">${t.priority === 'high' ? '🔥 ' : ''}${escapeHtml(t.title)}</div>

              <div class="meta">📅 ${t.date}${t.memo ? ' · ' + escapeHtml(t.memo) : ''}</div>

            </div>

            <div class="tools">

              <button class="btn" data-act="edit" data-id="${t.id}">수정</button>

              <button class="btn danger" data-act="delete" data-id="${t.id}">삭제</button>

            </div>

          </div>`).join('');

    }).join('') || `<div class="empty">선택한 담당자의 할 일이 없습니다.</div>`;

  }


  function escapeHtml(s) {

    return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

  }


  function renderAll() { renderDday(); renderCalendar(); renderPanel(); }


  // 폼: 생성 / 수정

  function resetForm() {

    editingId = null;

    $('todoForm').reset();

    $('date').value = selectedDate;

    $('submitBtn').textContent = '추가';

    $('cancelEdit').style.display = 'none';

  }


  $('todoForm').addEventListener('submit', (e) => {

    e.preventDefault();

    const data = {

      title: $('title').value.trim(),

      owner: document.querySelector('input[name=owner]:checked').value,

      date: $('date').value,

      priority: $('priority').value,

      memo: $('memo').value.trim(),

    };

    if (!data.title || !data.date) return;


    if (editingId) {

      const t = todos.find(x => x.id === editingId);

      if (t) Object.assign(t, data);

    } else {

      todos.push({ id: uid(), done: false, ...data });

    }

    save();

    // 추가한 날짜가 다른 달이면 해당 달로 이동

    const d = new Date(data.date + 'T00:00:00');

    selectedDate = data.date;

    viewYear = d.getFullYear(); viewMonth = d.getMonth();

    resetForm();

    renderAll();

  });


  $('cancelEdit').addEventListener('click', resetForm);


  // 할 일 목록 액션 (이벤트 위임)

  $('todoList').addEventListener('click', (e) => {

    const btn = e.target.closest('[data-act]');

    if (!btn) return;

    const id = btn.dataset.id;

    const t = todos.find(x => x.id === id);

    if (!t) return;

    const act = btn.dataset.act;


    if (act === 'delete') {

      if (!confirm('이 할 일을 삭제할까요?')) return;

      todos = todos.filter(x => x.id !== id);

      save();

      if (editingId === id) resetForm();

      renderAll();

    }

    if (act === 'edit') {

      editingId = id;

      $('title').value = t.title;

      document.querySelector(`input[name=owner][value=${t.owner}]`).checked = true;

      $('date').value = t.date;

      $('priority').value = t.priority;

      $('memo').value = t.memo || '';

      $('submitBtn').textContent = '저장';

      $('cancelEdit').style.display = '';

      $('title').focus();

    }

  });


  $('todoList').addEventListener('change', (e) => {

    if (e.target.dataset.act === 'toggle') {

      const t = todos.find(x => x.id === e.target.dataset.id);

      if (t) { t.done = e.target.checked; save(); renderAll(); }

    }

  });


  // 캘린더 상호작용

  $('calGrid').addEventListener('click', (e) => {

    const cell = e.target.closest('.day');

    if (!cell) return;

    selectedDate = cell.dataset.date;

    const d = new Date(selectedDate + 'T00:00:00');

    if (d.getMonth() !== viewMonth || d.getFullYear() !== viewYear) {

      viewYear = d.getFullYear(); viewMonth = d.getMonth();

    }

    if (!editingId) $('date').value = selectedDate;

    renderAll();

  });


  $('prev').onclick = () => { viewMonth--; if (viewMonth < 0) { viewMonth = 11; viewYear--; } renderAll(); };

  $('next').onclick = () => { viewMonth++; if (viewMonth > 11) { viewMonth = 0; viewYear++; } renderAll(); };

  $('today').onclick = () => {

    const now = new Date();

    viewYear = now.getFullYear(); viewMonth = now.getMonth();

    selectedDate = fmt(now);

    if (!editingId) $('date').value = selectedDate;

    renderAll();

  };


  document.querySelector('.filters').addEventListener('click', (e) => {

    const b = e.target.closest('button');

    if (!b) return;

    filter = b.dataset.filter;

    renderPanel();

  });


  // 디자인 시안 전환

  const savedTheme = localStorage.getItem('wedding-theme') || 'a';

  function applyTheme(t) {

    document.body.className = t === 'a' ? '' : 'theme-' + t;

    $('themeSelect').value = t;

    localStorage.setItem('wedding-theme', t);

  }

  $('themeSelect').addEventListener('change', (e) => applyTheme(e.target.value));

  applyTheme(savedTheme);


  // 초기화

  $('date').value = selectedDate;

  renderAll();

})();

</script>
